static Widget list({
  required BuildContext context,
  required int count,
  required Widget Function() builder,
}) {
  return Column(
    children: List.generate(count, (i) {
      return Padding(
        padding: EdgeInsets.only(bottom: i < count - 1 ? 8 : 0),
        child: builder(),
      );
    }),
  );
}
